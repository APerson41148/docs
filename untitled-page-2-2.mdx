Missle Mania is a fun 3rd person shooter game in which you use a variety of weapons to blast away enemies. It has a somewhat retro-y feel, and is extremely modifiable/customizable.

```bash
import pygame
import random
import sys
import math

pygame.init()

WIDTH, HEIGHT = 1440, 910
SCREEN = pygame.display.set_mode((WIDTH, HEIGHT), pygame.FULLSCREEN)
pygame.display.set_caption("Missle Mania")
CLOCK = pygame.time.Clock()
FPS = 60

COLOR_BACKGROUND = (0, 0, 0)
COLOR_PLAYER = (0, 255, 0)
COLOR_ENEMY = (255, 0, 0)
COLOR_ENEMY_BLUE = (0, 150, 255)
COLOR_TEXT = (255, 255, 255)
COLOR_MISSILE_OUTER = (0, 255, 255)
COLOR_MISSILE_INNER = (255, 255, 0)

PLAYER_SIZE, ENEMY_SIZE, PLAYER_SPEED = 30, 30, 5
ENEMY_SPEED_MIN, ENEMY_SPEED_MAX, NUM_ENEMIES = 3, 6, 5
RADAR_BOX_SIZE = 160
HEAT_SEEKING_COOLDOWN = 3000
MINIGUN_FIRE_RATE = 80


class Player:
    def __init__(self):
        self.rect = pygame.Rect(WIDTH//2 - 15, HEIGHT//2 - 15, PLAYER_SIZE, PLAYER_SIZE)

    def update(self, keys):
        dx = (keys[pygame.K_d] - keys[pygame.K_a]) * PLAYER_SPEED
        dy = (keys[pygame.K_s] - keys[pygame.K_w]) * PLAYER_SPEED
        self.rect.x += dx
        self.rect.y += dy
        self.rect.clamp_ip(pygame.Rect(0, 0, WIDTH, HEIGHT))

    def draw(self, surface):
        pygame.draw.rect(surface, COLOR_PLAYER, self.rect)


class Enemy:
    def __init__(self, player_rect):
        self.rect = pygame.Rect(0, 0, ENEMY_SIZE, ENEMY_SIZE)
        self.spawn_safely(player_rect)
        self.dx = random.choice([-1, 1]) * random.uniform(ENEMY_SPEED_MIN, ENEMY_SPEED_MAX)
        self.dy = random.choice([-1, 1]) * random.uniform(ENEMY_SPEED_MIN, ENEMY_SPEED_MAX)
        
        self.is_blue = (random.random() < 0.20)
        self.color = COLOR_ENEMY_BLUE if self.is_blue else COLOR_ENEMY
        
        self.last_shot_time = pygame.time.get_ticks() + random.randint(0, 1500)
        self.shoot_cooldown = 2500

    def spawn_safely(self, player_rect):
        while True:
            self.rect.x = random.randint(0, WIDTH - ENEMY_SIZE)
            self.rect.y = random.randint(0, HEIGHT - ENEMY_SIZE)
            dist = math.hypot(self.rect.centerx - player_rect.centerx, self.rect.centery - player_rect.centery)
            if dist > 200:
                break

    def update(self, mult):
        self.rect.x += self.dx * mult
        self.rect.y += self.dy * mult
        if self.rect.left <= 0 or self.rect.right >= WIDTH: 
            self.dx *= -1
        if self.rect.top <= 0 or self.rect.bottom >= HEIGHT: 
            self.dy *= -1
 
    def draw(self, surface):
        pygame.draw.rect(surface, self.color, self.rect)


class BaseProjectile:
    def __init__(self, x, y, w, h):
        self.x, self.y = float(x), float(y)
        self.rect = pygame.Rect(0, 0, w, h)
        self.rect.center = (int(self.x), int(self.y))
        self.active = True
        self.owner = "player"

    def update(self, enemies, player): pass
    def is_expired(self): return not self.active
    def is_off_screen(self):
        return self.rect.right < 0 or self.rect.left > WIDTH or self.rect.bottom < 0 or self.rect.top > HEIGHT

    def on_hit(self, enemy, enemies, timers):
        if enemy in enemies:
            enemies.remove(enemy)
            timers.append(pygame.time.get_ticks() + 6000)
        self.active = False

    def draw(self, surface): pass


class HomingMissile(BaseProjectile):
    def __init__(self, pos, mult=1.0):
        super().__init__(pos[0], pos[1], 16, 16)
        self.speed = 8.0 * mult
        self.spawn_time = pygame.time.get_ticks()
        
        mx, my = pygame.mouse.get_pos()
        dx, dy = mx - self.x, my - self.y
        dist = math.hypot(dx, dy)
        self.vx = (dx / dist * self.speed) if dist > 0 else 0
        self.vy = (dy / dist * self.speed) if dist > 0 else -self.speed

    def update(self, enemies, player):
        mx, my = pygame.mouse.get_pos()
        dx, dy = mx - self.x, my - self.y
        dist = math.hypot(dx, dy)
        if dist > 0:
            self.vx += (dx / dist * self.speed - self.vx) * 0.1
            self.vy += (dy / dist * self.speed - self.vy) * 0.1
        self.x += self.vx
        self.y += self.vy
        self.rect.center = (int(self.x), int(self.y))

    def is_expired(self):
        return not self.active or (pygame.time.get_ticks() - self.spawn_time >= 5000)

    def draw(self, surface):
        pygame.draw.circle(surface, COLOR_MISSILE_OUTER, self.rect.center, 10, 2)
        pygame.draw.circle(surface, COLOR_MISSILE_INNER, self.rect.center, 4)


class RadarMissile(BaseProjectile):
    def __init__(self, pos, enemy, mult=1.0):
        super().__init__(pos[0], pos[1], 12, 12)
        self.target, self.last_pos = enemy, enemy.rect.center
        self.speed = 12.0 * mult
        self.crazy = random.random() < 0.25
        mx, my = pygame.mouse.get_pos()
        dx, dy = mx - self.x, my - self.y
        dist = math.hypot(dx, dy)
        self.vx = (dx / dist * self.speed) if dist > 0 else 0
        self.vy = (dy / dist * self.speed) if dist > 0 else -self.speed

    def update(self, enemies, player):
        if self.crazy:
            self.vx += random.uniform(-1.5, 1.5)
            self.vy += random.uniform(-1.5, 1.5)
            dist = math.hypot(self.vx, self.vy)
            if dist > 0:
                self.vx, self.vy = self.vx / dist * self.speed, self.vy / dist * self.speed
        else:
            if self.target in enemies: 
                self.last_pos = self.target.rect.center
            dx, dy = self.last_pos[0] - self.x, self.last_pos[1] - self.y
            dist = math.hypot(dx, dy)
            if dist > 0:
                self.vx, self.vy = dx / dist * self.speed, dy / dist * self.speed
            else:
                self.vx, self.vy = 0, 0
        self.x += self.vx
        self.y += self.vy
        self.rect.center = (int(self.x), int(self.y))
        if not self.crazy:
            if math.hypot(self.last_pos[0] - self.x, self.last_pos[1] - self.y) < 8 and self.target not in enemies:
                self.active = False

    def draw(self, surface):
        pygame.draw.circle(surface, (255, 165, 0) if self.crazy else (0, 255, 0), self.rect.center, 8, 2)
        pygame.draw.circle(surface, (255, 0, 0), self.rect.center, 3)


class HeatSeekingMissile(BaseProjectile):
    def __init__(self, pos, enemies, mult=1.0):
        super().__init__(pos[0], pos[1], 14, 14)
        self.speed = 7.0 * mult
        self.vx, self.vy, self.target = 0.0, -self.speed, None
        self.retarget(enemies)

    def retarget(self, enemies):
        closest, min_d = None, float('inf')
        for e in enemies:
            d = math.hypot(e.rect.centerx - self.x, e.rect.centery - self.y)
            if d < min_d: 
                closest, min_d = e, d
        self.target = closest

    def update(self, enemies, player):
        if self.target not in enemies: 
            self.retarget(enemies)
        if self.target in enemies:
            dx, dy = self.target.rect.centerx - self.x, self.target.rect.centery - self.y
            dist = math.hypot(dx, dy)
            if dist > 0:
                self.vx += (dx / dist * self.speed - self.vx) * 0.12
                self.vy += (dy / dist * self.speed - self.vy) * 0.12
        self.x += self.vx
        self.y += self.vy
        self.rect.center = (int(self.x), int(self.y))

    def on_hit(self, enemy, enemies, timers):
        if random.random() < 0.95 and enemy in enemies:
            enemies.remove(enemy)
            timers.append(pygame.time.get_ticks() + 6000)
        self.active = False

    def draw(self, surface):
        pygame.draw.circle(surface, (255, 140, 0), self.rect.center, 9, 2)
        pygame.draw.circle(surface, (255, 0, 128), self.rect.center, 4)


class MinigunBullet(BaseProjectile):

    def on_hit(self, enemy, enemies, timers):

            enemies.remove(enemy)
            timers.append(pygame.time.get_ticks() + 6000)
        self.active = False

    def draw(self, surface):

class NovaBullet(BaseProjectile):
    def __init__(self, pos, rad, mult=1.0):
        super().__init__(pos[0], pos[1], 10, 10)
        self.speed = 9.0 * mult
        self.spawn_time = pygame.time.get_ticks()
        self.vx, self.vy = math.cos(rad) * self.speed, math.sin(rad) * self.speed

    def update(self, enemies, player):
        self.x += self.vx
        self.y += self.vy
        self.rect.center = (int(self.x), int(self.y))

        if self.rect.left <= 0:
            self.rect.left = 0
            self.vx *= -1
            self.x = float(self.rect.centerx)
        elif self.rect.right >= WIDTH:
            self.rect.right = WIDTH
            self.vx *= -1
            self.x = float(self.rect.centerx)

        if self.rect.top <= 0:
            self.rect.top = 0
            self.vy *= -1
            self.y = float(self.rect.centery)
        elif self.rect.bottom >= HEIGHT:
            self.rect.bottom = HEIGHT
            self.vy *= -1
            self.y = float(self.rect.centery)

    def is_expired(self):
        return not self.active or (pygame.time.get_ticks() - self.spawn_time >= 4000)

    def is_off_screen(self):
        return False

    def draw(self, surface):
        pygame.draw.circle(surface, (210, 0, 255), self.rect.center, 6)
        pygame.draw.circle(surface, (255, 255, 255), self.rect.center, 2)

class Slider:
    def __init__(self, x, y, width, min_val, max_val, start_val, label="Difficulty"):
        self.rect = pygame.Rect(x, y, width, 8)
        self.min_val = min_val
        self.max_val = max_val
        self.val = start_val
        self.label = label
        self.knob_radius = 8
        self.knob_x = self.val_to_x(start_val)
        self.dragging = False

    def val_to_x(self, val):
        ratio = (val - self.min_val) / (self.max_val - self.min_val)
        return self.rect.x + ratio * self.rect.width

    def x_to_val(self, x):
        ratio = (x - self.rect.x) / self.rect.width
        ratio = max(0.0, min(1.0, ratio))
        return self.min_val + ratio * (self.max_val - self.min_val)

    def update(self, event):
        if event.type == pygame.MOUSEBUTTONDOWN and event.button == 1:
            click_area = self.rect.inflate(0, 12)
            if click_area.collidepoint(event.pos):
                self.dragging = True
                self.knob_x = max(self.rect.left, min(event.pos[0], self.rect.right))
                self.val = self.x_to_val(self.knob_x)
        elif event.type == pygame.MOUSEBUTTONUP and event.button == 1:
            self.dragging = False
        elif event.type == pygame.MOUSEMOTION and self.dragging:
            self.knob_x = max(self.rect.left, min(event.pos[0], self.rect.right))
            self.val = self.x_to_val(self.knob_x)

    def draw(self, surface, font):
        pygame.draw.rect(surface, (80, 80, 80), self.rect, border_radius=4)
        progress_width = self.knob_x - self.rect.x
        if progress_width > 0:
            progress_rect = pygame.Rect(self.rect.x, self.rect.y, progress_width, self.rect.height)
            pygame.draw.rect(surface, (0, 150, 255), progress_rect, border_radius=4)
        pygame.draw.circle(surface, COLOR_TEXT, (int(self.knob_x), self.rect.centery), self.knob_radius)
        text_surf = font.render(f"{self.label}: {self.val:.1f}x", True, COLOR_TEXT)
        surface.blit(text_surf, (self.rect.x, self.rect.y - 20))


def draw_hotbar(surface, selected_slot, font):
    num_slots = len(WEAPONS)
    slot_size, gap = 36, 6
    start_x = (WIDTH - (num_slots * (slot_size + gap))) // 2
    y = HEIGHT - 50
    for i in range(1, num_slots + 1):
        rect = pygame.Rect(start_x + (i - 1) * (slot_size + gap), y, slot_size, slot_size)
        is_selected = (selected_slot == i)
        bg_color = (40, 40, 40) if not is_selected else (0, 100, 200)
        border_color = (150, 150, 150) if not is_selected else (0, 255, 255)
        
        pygame.draw.rect(surface, bg_color, rect, border_radius=4)
        pygame.draw.rect(surface, border_color, rect, 2, border_radius=4)
        
        num_text = font.render(str(i), True, (200, 200, 200))
        surface.blit(num_text, (rect.x + 4, rect.y + 2))
        
        abbrev = WEAPONS[i]["name"]
        content_text = font.render(abbrev, True, COLOR_TEXT)
        content_rect = content_text.get_rect(center=(rect.centerx, rect.centery + 8))
        surface.blit(content_text, content_rect)


def draw_cooldown_bar(surface, last_fire_time, cooldown_duration, weapon_name, font):
    current_time = pygame.time.get_ticks()
    elapsed = current_time - last_fire_time
    ratio = min(1.0, elapsed / cooldown_duration)
    if ratio < 1.0:
        bar_width, bar_height = 150, 8
        x = (WIDTH - bar_width) // 2
        y = HEIGHT - 70
        pygame.draw.rect(surface, (80, 80, 80), (x, y, bar_width, bar_height), border_radius=4)
        pygame.draw.rect(surface, (255, 140, 0), (x, y, int(bar_width * ratio), bar_height), border_radius=4)
        
        text_surf = font.render(f"{weapon_name} CD", True, (255, 140, 0))
        text_rect = text_surf.get_rect(center=(WIDTH // 2, y - 10))
        surface.blit(text_surf, text_rect)


def draw_game_over_screen(surface):

def main():
    ui_font = pygame.font.SysFont(None, 20)
    player = Player()
    difficulty_slider = Slider(20, 35, 150, 0.5, 2.5, 1.0)
    selected_slot = 1
    
    enemies = [Enemy(player.rect) for _ in range(NUM_ENEMIES)]
    active_projectiles = []  
    dead_enemies_timers = [] 
    locked_targets = []
    last_fire_times = {slot: 0 for slot in WEAPONS}
    game_over = False

    while True:
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                pygame.quit()
                sys.exit()
            
            difficulty_slider.update(event)
            
            if event.type == pygame.MOUSEBUTTONDOWN and event.button == 1:
                if not difficulty_slider.rect.inflate(0, 12).collidepoint(event.pos):
                    if selected_slot == 2 and not game_over:
                        mx, my = event.pos
                        radar_rect = pygame.Rect(mx - RADAR_BOX_SIZE // 2, my - RADAR_BOX_SIZE // 2, RADAR_BOX_SIZE, RADAR_BOX_SIZE)
                        for enemy in enemies:
                            if radar_rect.colliderect(enemy.rect) and enemy not in locked_targets:
                                locked_targets.append(enemy)

            if event.type == pygame.KEYDOWN:
                if event.key == pygame.K_f:
                    pygame.quit()
                    sys.exit()

                key_bindings = {
                    pygame.K_1: 1, pygame.K_2: 2, pygame.K_3: 3, pygame.K_4: 4, pygame.K_5: 5,
                    pygame.K_6: 6, pygame.K_7: 7, pygame.K_8: 8, pygame.K_9: 9, pygame.K_0: 10
                }
                if event.key in key_bindings:
                    slot = key_bindings[event.key]
                    if slot in WEAPONS: 
                        selected_slot = slot
                
                elif event.key == pygame.K_q:
                    selected_slot = selected_slot - 1 if selected_slot > 1 else len(WEAPONS)
                elif event.key == pygame.K_e:
                    selected_slot = selected_slot + 1 if selected_slot < len(WEAPONS) else 1
                
                if not game_over:
                    if event.key == pygame.K_SPACE:
                        weapon = WEAPONS[selected_slot]
                        if weapon["type"] == "click":
                            current_time = pygame.time.get_ticks()
                            if current_time - last_fire_times[selected_slot] >= weapon["cooldown"]:
                                fire_args = {
                                    "projs": active_projectiles,
                                    "p": player,
                                    "e": enemies,
                                    "locked": locked_targets,
                                    "diff": difficulty_slider.val
                                }
                                weapon["fire"](fire_args)
                                last_fire_times[selected_slot] = current_time
                else:
                    if event.key == pygame.K_r:
                        player = Player()
                        active_projectiles.clear()
                        enemies = [Enemy(player.rect) for _ in range(NUM_ENEMIES)]
                        dead_enemies_timers.clear()
                        locked_targets.clear()
                        selected_slot = 1
                        last_fire_times = {slot: 0 for slot in WEAPONS}
                        game_over = False

        hovered_enemies = []
        if not game_over:
            keys = pygame.key.get_pressed()
            player.update(keys)
            
            weapon = WEAPONS[selected_slot]
            if keys[pygame.K_SPACE] and weapon["type"] == "hold":
                current_time = pygame.time.get_ticks()
                if current_time - last_fire_times[selected_slot] >= weapon["cooldown"]:
                    fire_args = {
                        "projs": active_projectiles,
                        "p": player,
                        "e": enemies,
                        "locked": locked_targets,
                        "diff": difficulty_slider.val
                    }
                    weapon["fire"](fire_args)
                    last_fire_times[selected_slot] = current_time

            if selected_slot == 2:
                mx, my = pygame.mouse.get_pos()
                radar_rect = pygame.Rect(mx - RADAR_BOX_SIZE // 2, my - RADAR_BOX_SIZE // 2, RADAR_BOX_SIZE, RADAR_BOX_SIZE)
                for enemy in enemies:
                    if radar_rect.colliderect(enemy.rect):
                        hovered_enemies.append(enemy)
            
            for proj in list(active_projectiles):
                proj.update(enemies, player)
                if proj.is_off_screen() or proj.is_expired():
                    if proj in active_projectiles: 
                        active_projectiles.remove(proj)

            current_time = pygame.time.get_ticks()
            for respawn_time in list(dead_enemies_timers):
                if current_time >= respawn_time:
                    enemies.append(Enemy(player.rect))
                    dead_enemies_timers.remove(respawn_time)

            for enemy in list(enemies):
                enemy.update(difficulty_slider.val)
                
                if enemy.is_blue:
                    current_time = pygame.time.get_ticks()
                    if current_time - enemy.last_shot_time >= enemy.shoot_cooldown:
                        active_projectiles.append(EnemyMissile(enemy.rect.center, player.rect.center, difficulty_slider.val))
                        enemy.last_shot_time = current_time

                if player.rect.colliderect(enemy.rect):
                    game_over = True
                    
            for proj in list(active_projectiles):
                if proj.owner == "player":
                    for enemy in list(enemies):
                        if proj.rect.colliderect(enemy.rect):
                            proj.on_hit(enemy, enemies, dead_enemies_timers)
                            if enemy not in enemies and enemy in locked_targets:
                                locked_targets.remove(enemy)
                elif proj.owner == "enemy":
                    if proj.rect.colliderect(player.rect):
                        game_over = True
                        proj.active = False

        SCREEN.fill(COLOR_BACKGROUND)
        
        player.draw(SCREEN)
        for enemy in enemies: 
            enemy.draw(SCREEN)
        for proj in active_projectiles: 
            proj.draw(SCREEN)
        
        if not game_over:
            if selected_slot == 2:
                mx, my = pygame.mouse.get_pos()
                radar_rect = pygame.Rect(mx - RADAR_BOX_SIZE // 2, my - RADAR_BOX_SIZE // 2, RADAR_BOX_SIZE, RADAR_BOX_SIZE)
                pygame.draw.rect(SCREEN, (0, 255, 0), radar_rect, 1)
                pygame.draw.line(SCREEN, (0, 255, 0), (mx - 6, my), (mx + 6, my), 1)
```
